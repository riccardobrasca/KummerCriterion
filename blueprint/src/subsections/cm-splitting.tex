% L4: CM field, class number splitting, proof that h^+ | h.

\section{The maximal real subfield}

\begin{definition}\label{def:cm-real-subfield}\lean{NumberField.maximalRealSubfield}\leanok
The \emph{maximal real subfield} of $\Qzetap$ is
\[
  \Qzetapplus \;=\; \Q(\zetap + \zetap^{-1}),
\]
the fixed field of complex conjugation. It is a totally real subfield of
$\Qzetap$ of degree $(p-1)/2$ over $\Q$, and $\Qzetap$ is a CM field over
$\Qzetapplus$.
\end{definition}

The class-number splitting begins by separating the class number of the full
cyclotomic field from that of its maximal real subfield.

\section[The class numbers h, h+, and h-]{The class numbers $h$, $h^+$, and $h^-$}

\begin{definition}\label{def:class-number-h}\lean{KummerCriterion.h}\leanok
The class number of $\Qzetap$ is
\[
  \hp \;:=\; \abs{\Cl{\OK[\Qzetap]}}.
\]
\end{definition}

\begin{definition}\label{def:class-number-hplus}\lean{KummerCriterion.hPlus}\leanok
The \emph{plus class number} is the class number of the maximal real subfield:
\[
  \hplus \;:=\; \abs{\Cl{\OK[\Qzetapplus]}}.
\]
\end{definition}

The crucial foundational fact is that $\hplus$ divides $\hp$.  Only after this
has been proved can one define the relative class number by honest division.
The proof, occupying the rest of this chapter, follows Diekmann's
Proposition~55 (Washington's Theorem~4.14).  The strategy is to show that the
natural extension-of-ideals map
\[
  \Cl{\OK[\Qzetapplus]} \longrightarrow \Cl{\OK[\Qzetap]}
\]
is injective; divisibility of the orders then follows from the fact that an
injective homomorphism of finite groups has image of order dividing the
codomain.

\section{The class-group map and faithful-flat descent}

There is a natural extension-of-ideals map
\[
  \Cl{\OK[\Qzetapplus]} \longrightarrow \Cl{\OK[\Qzetap]},
  \qquad [I] \longmapsto [I\,\mathcal O_{\Q(\zeta_p)}].
\]
Throughout this chapter we abbreviate
$\mathcal O := \mathcal O_{\Q(\zeta_p)}$ and
$\mathcal O^+ := \mathcal O_{\Q(\zeta_p)^+}$.

The proof that $\hplus \mid \hp$ reduces, by injectivity-implies-divisibility
on finite groups, to the statement that an ideal $I$ of $\mathcal O^+$ whose
extension $I \mathcal O$ is principal is itself principal.

The basic descent input is that the ring extension
$\mathcal O^+ \subseteq \mathcal O$ is well-behaved.

\begin{theorem}\label{thm:ring-of-integers-faithfully-flat}
  \lean{KummerCriterion.ringOfIntegers_faithfullyFlat_maximalRealSubfield}\leanok
  \uses{def:cm-real-subfield}
The ring of integers $\mathcal O_{\Q(\zeta_p)}$ is faithfully flat over
$\mathcal O_{\Q(\zeta_p)^+}$.
\end{theorem}

\begin{proof}\leanok
Flatness is automatic: $\mathcal O$ is finitely generated and projective as a
$\mathcal O^+$-module, because $\Qzetap / \Qzetapplus$ is a finite separable
extension and the rings of integers form a tower of Dedekind domains. To
upgrade flatness to faithful flatness it suffices, by the prime-spectrum
criterion, to show that the comap
$\operatorname{Spec} \mathcal O \to \operatorname{Spec} \mathcal O^+$
is surjective. Given a prime $\mathfrak q \subseteq \mathcal O^+$, choose any
prime $\mathfrak Q$ of $\mathcal O$ lying over $\mathfrak q$ (such a prime
exists by integrality of $\mathcal O$ over $\mathcal O^+$); then
$\mathfrak Q$ maps to $\mathfrak q$ under the comap.
\end{proof}

Faithful flatness gives the standard extension-contraction identity:

\begin{theorem}\label{thm:map-comap-ring-of-integers}
  \lean{KummerCriterion.map_comap_eq_ringOfIntegers}\leanok
  \uses{thm:ring-of-integers-faithfully-flat}
For every ideal $J \subseteq \mathcal O_{\Q(\zeta_p)^+}$ one has
\[
  (J\mathcal O_{\Q(\zeta_p)}) \cap \mathcal O_{\Q(\zeta_p)^+} = J.
\]
\end{theorem}

\begin{proof}\leanok
Faithful flatness of $\mathcal O$ over $\mathcal O^+$ is equivalent to the
statement that the canonical map $J \to J \mathcal O$ is injective for every
ideal $J$ of $\mathcal O^+$, and more strongly that contracting an extended
ideal recovers the original. Concretely, $J \mathcal O \cap \mathcal O^+
\subseteq J$ holds for any faithfully flat extension by the standard
descent fact; the reverse inclusion is automatic.
\end{proof}

Consequently, principality of an extended ideal descends as soon as one can
exhibit a generator that already lives in the smaller ring of integers.

\begin{theorem}\label{thm:is-principal-map-span-singleton}
  \lean{KummerCriterion.isPrincipal_of_map_eq_span_singleton_of_mem}\leanok
  \uses{thm:map-comap-ring-of-integers}
Let $I$ be an ideal of $\mathcal O_{\Q(\zeta_p)^+}$. If
\[
  I\mathcal O_{\Q(\zeta_p)} = (b)
\]
for some element $b$ lying in the image of
$\mathcal O_{\Q(\zeta_p)^+} \hookrightarrow \mathcal O_{\Q(\zeta_p)}$, say
$b = \iota(b_0)$ with $b_0 \in \mathcal O^+$, then $I = (b_0)$; in
particular $I$ is principal.
\end{theorem}

\begin{proof}\leanok
Write $\iota : \mathcal O^+ \hookrightarrow \mathcal O$ for the inclusion.
We use \cref{thm:map-comap-ring-of-integers} twice. First, applied to $I$,
\[
  I = \iota^{-1}(I \mathcal O) = \iota^{-1}((b)) = \iota^{-1}((\iota(b_0))).
\]
Second, applied to the principal ideal $(b_0) \subseteq \mathcal O^+$,
\[
  \iota^{-1}((\iota(b_0))) = \iota^{-1}((b_0) \mathcal O) = (b_0).
\]
Combining the two displays gives $I = (b_0)$.
\end{proof}

The remaining work is therefore to start from an arbitrary generator $a$ of
$I \mathcal O$ upstairs and replace it by an associate fixed by complex
conjugation; that associate will then lie in $\mathcal O^+$.

\section{Conjugation on principal generators}

Let $I$ be an ideal of $\mathcal O^+$ such that
\[
  I \mathcal O = (a)
\]
for some $a \in \mathcal O$, and write $\overline{\,\cdot\,}$ for the action
of complex conjugation on $\mathcal O$.  Because $I$ comes from the real
subfield, the extended ideal $I \mathcal O$ is fixed by conjugation.

\begin{theorem}\label{thm:ideal-map-conj}
  \lean{KummerCriterion.ideal_map_conj_eq}\leanok
  \uses{def:cm-real-subfield}
If $I$ is an ideal of $\mathcal O_{\Q(\zeta_p)^+}$, then the extended ideal
$I\mathcal O_{\Q(\zeta_p)}$ is fixed by complex conjugation.
\end{theorem}

\begin{proof}\leanok
Conjugation fixes every element of $\mathcal O^+$ by definition of the
maximal real subfield, so it commutes with the algebra map
$\mathcal O^+ \hookrightarrow \mathcal O$. Functoriality of ideal extension
gives
\[
  \overline{I \mathcal O} = \overline{(\iota I)} = \iota I = I \mathcal O,
\]
where $\iota$ denotes the inclusion.
\end{proof}

Hence $a$ and $\overline a$ generate the same principal ideal, so they differ
by a unit.

\begin{theorem}\label{thm:conj-generator-associated}
  \lean{KummerCriterion.conj_generator_associated}\leanok
  \uses{thm:ideal-map-conj}
There exists a unit $u \in \mathcal O_{\Q(\zeta_p)}^\times$ such that
\[
  \overline a = u\,a.
\]
\end{theorem}

\begin{proof}\leanok
From $I \mathcal O = (a)$ and $\overline{I \mathcal O} = I \mathcal O$ we have
$(a) = \overline{(a)} = (\overline a)$ in $\mathcal O$. Two generators of the
same nonzero principal ideal in an integral domain are associates, so
$\overline a = u a$ for some $u \in \mathcal O^\times$.
\end{proof}

Applying conjugation a second time recovers $a$, which forces the unit $u$ to
be antisymmetric under conjugation.

\begin{theorem}\label{thm:conj-unit-mul-one}
  \lean{KummerCriterion.conj_unit_mul_eq_one}\leanok
  \uses{thm:conj-generator-associated}
The unit $u$ satisfies
\[
  u\,\overline u = 1.
\]
\end{theorem}

\begin{proof}\leanok
Apply conjugation to $\overline a = u a$ and use $\overline{\overline a} = a$:
\[
  a = \overline{\overline a} = \overline u \cdot \overline a = \overline u \cdot u\,a.
\]
Cancelling $a \ne 0$ in the domain $\mathcal O$ gives $\overline u \, u = 1$.
\end{proof}

The next ingredient is the classification of antisymmetric units in a
cyclotomic field, sometimes called the Kronecker–Kummer lemma. We use $\zeta$
for $\zeta_p$ and $\langle \zeta \rangle$ for the cyclic group of $p$-th roots
of unity.

\begin{theorem}\label{thm:antisymmetric-unit-classification}
  \lean{KummerCriterion.antisymmetric_unit_eq_neg_one_pow_mul_zeta_pow}\leanok
  \uses{thm:conj-unit-mul-one}
If $u \in \mathcal O_{\Q(\zeta_p)}^\times$ satisfies $u\overline u = 1$, then
there exist integers $k,n$ such that
\[
  u = (-1)^k \zeta_p^n.
\]
\end{theorem}

\begin{proof}\leanok
Let $\sigma : K \hookrightarrow \C$ be any embedding. Since complex
conjugation on $K$ commutes with every embedding, the relation
$u\overline u = 1$ implies
\[
  |\sigma(u)|^2 = \sigma(u)\overline{\sigma(u)} = 1.
\]
Thus every Galois conjugate of $u$ has complex absolute value $1$.
Kronecker's theorem therefore shows that $u$ is a root of unity.

The roots of unity in $\Q(\zeta_p)$ are exactly the elements
\(
  \{\pm \zeta_p^n : 0 \le n \le p-1\},
\)
because $p$ is odd. Hence there exist integers $k,n$ such that
\[
  u = (-1)^k \zeta_p^n.
\]
\end{proof}

\section[A conjugation-fixed associate and descent to the real subfield ring of integers]{A conjugation-fixed associate and descent to $\mathcal O_{K^+}$}

The classification of the unit $u$ is used to modify the generator $a$ by an
associate so that the new generator is fixed by conjugation.  The serious
content is to eliminate the possibility $k$ odd; once this is done, $n$ can
be replaced by an even integer because $p$ is odd, and a $p$-th root of
unity adjustment produces the fixed associate.

\subsection*{Eliminating the \texorpdfstring{$-1$}{-1} factor}

Write $\zeta = \zeta_p$ and $\pi := \zeta - 1 \in \mathcal O$. Let
$P = (\pi)$ be the unique prime of $\mathcal O$ above $p$, and
$P^+ \subseteq \mathcal O^+$ the prime below it. The extension
$\mathcal O^+ \subseteq \mathcal O$ ramifies totally at $P^+$ with
$e(P / P^+) = 2$, that is, $P^+ \mathcal O = P^2$.

\begin{theorem}[Ramification at $\pi$]\label{thm:ramificationIdx-zetaPrimePlus}
  \lean{KummerCriterion.ramificationIdx_zetaPrimePlus_eq_two}\leanok
  \uses{def:cm-real-subfield}
The ramification index of $P^+$ in $\mathcal O$ equals $2$:
\[
  P^+ \mathcal O = P^2.
\]
\end{theorem}

\begin{proof}\leanok
Inside $\mathcal O$ one has the identity $P^+ \mathcal O = P^2$ by an
explicit computation with $\zeta + \zeta^{-1} - 2 = -\pi \overline\pi
\zeta^{-1}$, exhibiting the totally ramified extension at the unique prime
above $p$ in the maximal real subfield.
\end{proof}

\begin{lemma}[Even multiplicity of $\pi$ in a descended generator]
  \label{thm:multiplicity-zetaPrime-even}
  \lean{KummerCriterion.multiplicity_zetaPrime_even_of_map_eq_span}\leanok
  \uses{thm:ramificationIdx-zetaPrimePlus}
If $I$ is a nonzero ideal of $\mathcal O^+$ with
$I \mathcal O = (a)$ and $a \ne 0$, then $v_\pi(a)$ is even.
\end{lemma}

\begin{proof}\leanok
Let $e = v_\pi(a)$, equivalently the multiplicity with which $P = (\pi)$
appears in the prime factorisation of $(a) = I \mathcal O$ in $\mathcal O$.
For Dedekind extensions one has
\[
  v_P(I \mathcal O) \;=\; e(P/P^+) \cdot v_{P^+}(I),
\]
and the ramification index $e(P / P^+) = 2$ by
\cref{thm:ramificationIdx-zetaPrimePlus}. Therefore
$v_\pi(a) = 2 \cdot v_{P^+}(I)$ is even.
\end{proof}

The classification \cref{thm:antisymmetric-unit-classification} is now
refined to remove the possible $-1$:

\begin{theorem}[Pure $\zeta$-power form of the unit]
  \label{thm:generator-unit-eq-zeta-pow}
  \lean{KummerCriterion.generator_unit_eq_zeta_pow}\leanok
  \uses{thm:antisymmetric-unit-classification, thm:multiplicity-zetaPrime-even}
With $I, a, u$ as above and writing $u = (-1)^k \zeta^n$, the integer $k$ is
necessarily even, so there exists $n$ with $u = \zeta^n$.
\end{theorem}

\begin{proof}\leanok
Suppose for contradiction that $k$ is odd, so $u = -\zeta^n$. Let
$e = v_\pi(a)$ and write $a = \pi^e b$ with $\pi \nmid b$ (Lemma~\ref
{thm:multiplicity-zetaPrime-even} applies and $e$ is even). Conjugation
sends $\pi$ to $\overline\pi = \gamma \cdot \pi$, where the explicit unit
$\gamma$ can be computed as $\gamma = -\zeta^{p-1}$ from
$\overline\zeta = \zeta^{p-1}$. Substituting into $\overline a = u a$ and
cancelling $\pi^e$ — using that $\pi$ is prime and that $\gamma^e \cdot
\zeta^e = ((-1) \zeta)^e \cdot \zeta^{(p-1)e - e}$ simplifies using
$e$ even — yields
\[
  \overline b \;=\; u' \cdot b
  \qquad\text{where}\qquad
  u' \;:=\; \zeta^e \cdot u \;=\; -\zeta^{e+n}.
\]
Reducing modulo the prime $P = (\pi)$ kills $\pi$ and any factor of the
form $\zeta^m - 1$, so $\zeta^{e+n} \equiv 1 \pmod P$ and therefore
$u' + 1 \equiv 0 \pmod P$.  On the other hand $\overline b \equiv b
\pmod P$ since conjugation is trivial modulo $\pi$, so
$b (u' - 1) \equiv 0 \pmod P$.  Since $b \notin P$, this forces
$u' - 1 \in P$.  Combining $u' + 1 \in P$ and $u' - 1 \in P$ gives
$2 \in P$, which contradicts $p$ odd (the prime $P$ lies above $p$, not
$2$).  Therefore $k$ must be even, and $(-1)^k = 1$.
\end{proof}

\subsection*{Constructing the conjugation-fixed associate}

\begin{theorem}\label{thm:conj-fixed-associate}
  \lean{KummerCriterion.exists_conj_fixed_associate_of_classification}\leanok
  \uses{thm:generator-unit-eq-zeta-pow}
There exists $b \in \mathcal O_{\Q(\zeta_p)}$ such that
\[
  (b) = (a)
  \qquad\text{and}\qquad
  \overline b = b.
\]
\end{theorem}

\begin{proof}\leanok
By \cref{thm:generator-unit-eq-zeta-pow} write $u = \zeta^n$. Since $p$ is
odd, $2$ is invertible modulo $p$, and the map $m \mapsto 2m$ is a bijection
on $\Z/p\Z$. Choose $m$ with $n \equiv 2m \pmod p$, so $\zeta^n =
(\zeta^m)^2$. Set
\[
  b \;:=\; a \cdot \zeta^m.
\]
Then $(b) = (a)$ because $\zeta^m \in \mathcal O^\times$, and using
$\overline{\zeta^m} = \zeta^{-m}$ together with
$\overline a = \zeta^n a = \zeta^{2m} a$,
\[
  \overline b \;=\; \overline a \cdot \zeta^{-m}
            \;=\; \zeta^{2m} a \cdot \zeta^{-m}
            \;=\; a \cdot \zeta^m \;=\; b.
\]
\end{proof}

An algebraic integer of $\mathcal O$ fixed by complex conjugation lies in the
image of $\mathcal O^+$.

\begin{theorem}\label{thm:mem-ring-of-integers-conj-self}
  \lean{KummerCriterion.mem_ringOfIntegers_of_conj_eq_self}\leanok
  \uses{thm:conj-fixed-associate}
If $b \in \mathcal O_{\Q(\zeta_p)}$ satisfies $\overline b = b$, then
$b$ lies in the image of
$\mathcal O_{\Q(\zeta_p)^+} \hookrightarrow \mathcal O_{\Q(\zeta_p)}$.
\end{theorem}

\begin{proof}\leanok
The maximal real subfield is by definition the fixed field of complex
conjugation: $\Qzetapplus = \Qzetap^{\overline{\,\cdot\,}}$. An element of
$\mathcal O \subseteq \Qzetap$ fixed by $\overline{\,\cdot\,}$ therefore lies
in $\Qzetapplus$, and being an algebraic integer it lies in $\mathcal O^+$.
\end{proof}

Combining this with faithful-flat descent gives principality downstairs.

\begin{theorem}[Diekmann Proposition 55 / Washington 4.14]
  \label{thm:is-principal-after-extension}
  \lean{KummerCriterion.isPrincipal_of_isPrincipal_map_Kplus}\leanok
  \uses{thm:is-principal-map-span-singleton, thm:mem-ring-of-integers-conj-self,
        thm:conj-fixed-associate}
Let $I$ be an ideal of $\mathcal O_{\Q(\zeta_p)^+}$. If the extended ideal
$I\mathcal O_{\Q(\zeta_p)}$ is principal, then $I$ itself is principal.
\end{theorem}

\begin{proof}\leanok
Write $I \mathcal O = (a)$. We may assume $a \ne 0$, since the case $I = 0$
is trivial.

By \cref{thm:ideal-map-conj} the ideal $I \mathcal O$ is conjugation-fixed,
so by \cref{thm:conj-generator-associated} there is a unit $u$ with
$\overline a = u a$, and by \cref{thm:conj-unit-mul-one} this unit satisfies
$u \overline u = 1$.  The classification of antisymmetric units
(\cref{thm:antisymmetric-unit-classification}) gives $u = (-1)^k \zeta^n$,
and \cref{thm:generator-unit-eq-zeta-pow} eliminates the $-1$, so
$u = \zeta^n$ for some $n$.

\Cref{thm:conj-fixed-associate} produces $b \in \mathcal O$ with $(b) = (a)$
and $\overline b = b$, and \cref{thm:mem-ring-of-integers-conj-self} gives
$b = \iota(b_0)$ for some $b_0 \in \mathcal O^+$.  We then have
$I \mathcal O = (b) = (\iota b_0)$, so
\cref{thm:is-principal-map-span-singleton} concludes that $I = (b_0)$ is
principal.
\end{proof}

\section[Injectivity of the class-group map and the definition of h-]{Injectivity of the class-group map and the definition of $h^-$}

The descent theorem immediately gives injectivity on class groups.

\begin{theorem}\label{thm:class-group-map-injective}
  \lean{KummerCriterion.classGroupMap_injective}\leanok
  \uses{thm:is-principal-after-extension}
The extension map
\[
  \Cl{\OK[\Qzetapplus]} \longrightarrow \Cl{\OK[\Qzetap]}
\]
is injective.
\end{theorem}

\begin{proof}\leanok
A homomorphism of groups is injective iff its kernel is trivial. An element
$[I] \in \Cl{\mathcal O^+}$ goes to the identity in $\Cl{\mathcal O}$ exactly
when $I \mathcal O$ is principal in $\mathcal O$.
\Cref{thm:is-principal-after-extension} says that this forces $I$ itself to
be principal, i.e.\ $[I] = 1$ in $\Cl{\mathcal O^+}$. Thus the kernel is
trivial.
\end{proof}

Since both class groups are finite, injectivity implies divisibility of their
cardinalities.

\begin{theorem}[$\hplus$ divides $\hp$]\label{thm:hplus-divides-h}
  \lean{KummerCriterion.hPlus_dvd_h}\leanok
  \uses{thm:class-group-map-injective, def:class-number-h, def:class-number-hplus}
The class number $\hplus$ of $\Qzetapplus$ divides the class number $\hp$ of
$\Qzetap$.
\end{theorem}

\begin{proof}\leanok
By \cref{thm:class-group-map-injective} the extension map $\Cl{\mathcal O^+}
\to \Cl{\mathcal O}$ is an injective homomorphism of finite groups. Its
image is a subgroup of $\Cl{\mathcal O}$ of order $\hplus$, and by Lagrange's
theorem the order of any subgroup of a finite group divides the order of
the ambient group.
\end{proof}

This justifies the definition of the relative class number.

\begin{definition}\label{def:class-number-hminus}\lean{KummerCriterion.hMinus}\leanok
  \uses{thm:hplus-divides-h}
The \emph{relative class number} is defined by
\[
  \hminus \;:=\; \hp / \hplus.
\]
\end{definition}

We shall also use the factorisation and positivity statements below.

\begin{theorem}\label{thm:h-factorisation}\lean{KummerCriterion.h_eq_hPlus_mul_hMinus}\leanok
  \uses{def:class-number-hminus, thm:hplus-divides-h}
One has
\[
  \hp = \hplus\,\hminus.
\]
\end{theorem}

\begin{proof}\leanok
Immediate from \cref{def:class-number-hminus} and
\cref{thm:hplus-divides-h}: dividing $\hp$ by the divisor $\hplus$ yields
$\hminus$, and multiplying back recovers $\hp$.
\end{proof}

\begin{theorem}\label{thm:hminus-positive}\lean{KummerCriterion.hMinus_pos}\leanok
  \uses{def:class-number-hminus, thm:hplus-divides-h}
The relative class number $\hminus$ is strictly positive.
\end{theorem}

\begin{proof}\leanok
Both $\hp$ and $\hplus$ are positive integers (cardinalities of nonempty
finite groups), and $\hplus \mid \hp$, so the quotient
$\hp / \hplus$ is a positive integer.
\end{proof}
