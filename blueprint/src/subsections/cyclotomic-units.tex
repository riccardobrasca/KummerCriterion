\section{The real cyclotomic-unit subgroup}

\begin{definition}\label{def:real-cyclotomic-unit}
  \lean{KummerCriterion.realCyclotomicUnit}\leanok
For $2\le a\le (p-1)/2$, the real cyclotomic unit is the unit of
$\mathcal O_{K^+}$ represented by
\[
  \frac{1-\zeta_p^a}{1-\zeta_p}.
\]
\end{definition}

\begin{definition}\label{def:CPlus-generator}
  \lean{KummerCriterion.CPlusGenerator}\leanok
  \uses{def:real-cyclotomic-unit}
The finite family of standard generators is indexed by
$a=2,\ldots,(p-1)/2$ and written as `CPlusGenerator`.
\end{definition}

\begin{definition}\label{def:CPlus}
  \lean{KummerCriterion.CPlus}\leanok
  \uses{def:CPlus-generator}
The subgroup $C^+\subseteq(\mathcal O_{K^+})^\times$ is generated by $-1$ and
the standard real cyclotomic units:
\[
  C^+=\langle -1,\varepsilon_2,\ldots,\varepsilon_{(p-1)/2}\rangle.
\]
\end{definition}

\begin{definition}\label{def:normalizedCPlus}
  \lean{KummerCriterion.normalizedCPlus}\leanok
  \uses{def:CPlus}
The normalized subgroup is generated by $-1$ and normalized square roots of the
standard real cyclotomic units.
\end{definition}

\begin{theorem}\label{thm:normalized-square}
  \lean{KummerCriterion.normalizedCPlusGenerator_sq_eq_CPlusGenerator}\leanok
  \uses{def:CPlus, def:normalizedCPlus}
Each normalized generator squares to the corresponding generator of $C^+$.
\end{theorem}
\begin{proof}\leanok
This is the normalization property built into the definition of the normalized
generators.
\end{proof}

\begin{theorem}\label{thm:CPlus-normalized-index}
  \lean{KummerCriterion.CPlus_index_prime_dvd_iff_normalizedCPlus_index_prime_dvd}\leanok
  \uses{thm:normalized-square}
For an odd prime $p$,
\[
  p\mid[(\mathcal O_{K^+})^\times:C^+]
  \Longleftrightarrow
  p\mid[(\mathcal O_{K^+})^\times:C^+_{\mathrm{norm}}].
\]
\end{theorem}
\begin{proof}\leanok
The relative quotient between $C^+$ and the normalized subgroup has order
dividing a power of $2$.  Since $p$ is odd, passing between the two indices
does not change $p$-divisibility.
\end{proof}

\section{\texorpdfstring{Exact $p$-saturation}{Exact p-saturation}}

\begin{definition}\label{def:EPlus}
  \lean{KummerCriterion.EPlus}\leanok
The ambient plus-side unit subgroup used for saturation is the full group
$(\mathcal O_{K^+})^\times$.
\end{definition}

\begin{definition}\label{def:pPowerSubgroup}
  \lean{KummerCriterion.pPowerSubgroup}\leanok
For a subgroup $H$ of an abelian group, $H^p$ denotes the exact image of the
$p$th-power map:
\[
  H^p=\{h^p:h\in H\}.
\]
\end{definition}

\begin{definition}\label{def:pSaturated}
  \lean{KummerCriterion.pSaturated}\leanok
  \uses{def:pPowerSubgroup}
A subgroup $H\le E$ is $p$-saturated in $E$ when
\[
  H\cap E^p\subseteq H^p.
\]
Equivalently, every element of $H$ that is a $p$th power in $E$ is already a
$p$th power inside $H$.
\end{definition}

\begin{definition}\label{def:CPlusExponentProduct}
  \lean{KummerCriterion.CPlusExponentProduct}\leanok
  \uses{def:CPlus}
An element written in the standard generators has the form
\[
  (-1)^s\prod_a\varepsilon_a^{e_a}.
\]
This expression is packaged with integer exponents $s,e_a$.
\end{definition}

\begin{theorem}\label{thm:exists-CPlusExponentProduct}
  \lean{KummerCriterion.exists_CPlusExponentProduct_of_mem_CPlus}\leanok
  \uses{def:CPlusExponentProduct}
Every element of $C^+$ admits such an integer-exponent expression.
\end{theorem}
\begin{proof}\leanok
This is induction on membership in the subgroup generated by $-1$ and the
finite family of real cyclotomic units.  The proof checks the generators,
closure under multiplication, inverses, and the identity element.
\end{proof}

\begin{theorem}\label{thm:CPlus-pSaturated-exponents}
  \lean{KummerCriterion.CPlus_pSaturated_of_generator_exponents_modP_zero}\leanok
  \uses{def:pSaturated, thm:exists-CPlusExponentProduct}
Assume that whenever
\[
  (-1)^s\prod_a\varepsilon_a^{e_a}
\]
is a $p$th power in the full unit group, every exponent $e_a$ is zero modulo
$p$.  Then $C^+$ is $p$-saturated in the full plus-side unit group.
\end{theorem}
\begin{proof}\leanok
Take an element of $C^+$ which is a $p$th power in the ambient unit group.
Write it as an exponent product using
\cref{thm:exists-CPlusExponentProduct}.  The hypothesis says each $e_a$ is
divisible by $p$.  Divide the exponents by $p$ to construct an element of
$C^+$ whose $p$th power is the original element.
\end{proof}

\begin{theorem}\label{thm:not-dvd-index-of-pSaturated}
  \lean{KummerCriterion.not_dvd_index_of_pSaturated}\leanok
  \uses{def:pSaturated, def:CPlus}
If $C^+$ is $p$-saturated in the full plus-side unit group, then
\[
  p\nmid[(\mathcal O_{K^+})^\times:C^+].
\]
\end{theorem}
\begin{proof}\leanok
The group-theoretic input says that a finite-index $p$-saturated subgroup has
index prime to $p$ when all ambient $p$-torsion already lies in the subgroup.
For $C^+$, the torsion condition follows from the description of torsion units
in $K^+$, and finite index is supplied by the cyclotomic-unit index setup.
\end{proof}

\section{The Kummer logarithm matrix}

\begin{definition}\label{def:kummerLogRank}
  \lean{KummerCriterion.CyclotomicUnits.kummerLogRank}\leanok
The Kummer logarithm matrix has rank $(p-3)/2$, matching the number of
standard real cyclotomic-unit generators.
\end{definition}

\begin{definition}\label{def:concreteKummerLogMatrix}
  \lean{KummerCriterion.CyclotomicUnits.concreteKummerLogMatrix}\leanok
  \uses{def:kummerLogRank}
The concrete Kummer logarithm matrix is the matrix over $\mathbb F_p$ whose
columns are the completed $p$-adic logarithms of the real cyclotomic-unit
generators, expressed in the Dwork basis.
\end{definition}

\begin{theorem}\label{thm:kummer-log-vandermonde}
  \lean{KummerCriterion.CyclotomicUnits.concreteKummerLogMatrix_eq_diagonal_mul_vandermonde}\leanok
  \uses{def:concreteKummerLogMatrix}
The concrete Kummer logarithm matrix factors as a diagonal matrix of
Bernoulli factors times a Teichmuller Vandermonde matrix.
\end{theorem}
\begin{proof}\leanok
The proof computes each logarithm coefficient.  The row-dependent part is the
Bernoulli factor, while the column-dependent part is a power of a Teichmuller
node.  This separates the matrix into a diagonal factor and a Vandermonde
factor.
\end{proof}

\begin{theorem}\label{thm:vandermonde-nonzero}
  \lean{KummerCriterion.CyclotomicUnits.vandermonde_teichmuller_even_sub_one_det_ne_zero}\leanok
The Teichmuller Vandermonde determinant occurring above is nonzero.
\end{theorem}
\begin{proof}\leanok
The Teichmuller nodes are distinct and no node is equal to $1$.  The
Vandermonde determinant is therefore a product of nonzero differences.
\end{proof}

\begin{theorem}\label{thm:kummer-det-bernoulli-factor}
  \lean{KummerCriterion.CyclotomicUnits.concreteKummerLogMatrix_det_ne_zero_iff_forall_bernoulliFactor_ne_zero}\leanok
  \uses{thm:kummer-log-vandermonde, thm:vandermonde-nonzero}
The Kummer logarithm determinant is nonzero exactly when every Bernoulli diagonal
factor is nonzero.
\end{theorem}
\begin{proof}\leanok
Use the diagonal-times-Vandermonde factorisation.  The Vandermonde determinant
is nonzero by \cref{thm:vandermonde-nonzero}, so the determinant is nonzero
exactly when every diagonal entry is nonzero.
\end{proof}

\begin{theorem}\label{thm:bernoulliFactor-num}
  \lean{KummerCriterion.CyclotomicUnits.bernoulliFactor_ne_zero_iff_not_dvd_bernoulli_num}\leanok
The diagonal Bernoulli factor attached to $j$ is nonzero modulo $p$ exactly
when
\[
  p\nmid\operatorname{num}(B_{2j}).
\]
\end{theorem}
\begin{proof}\leanok
The factor is the reduction modulo $p$ of a $p$-integral rational expression
whose unit part is the numerator of $B_{2j}$.  Denominator terms are units in
the permitted range.
\end{proof}

\begin{theorem}\label{thm:kummer-log-det-bernoulli}
  \lean{KummerCriterion.CyclotomicUnits.kummerLogMatrix_det_ne_zero_iff_bernoulli_nonzero}\leanok
  \uses{thm:kummer-det-bernoulli-factor, thm:bernoulliFactor-num}
The determinant is nonzero precisely when every Bernoulli numerator in
Kummer's range is nonzero modulo $p$:
\[
  \det M\ne0
  \Longleftrightarrow
  \forall j,\ 1\le j,\ 2j\le p-3,\quad
  p\nmid\operatorname{num}(B_{2j}).
\]
\end{theorem}
\begin{proof}\leanok
\Cref{thm:kummer-det-bernoulli-factor} reduces the determinant condition to
the diagonal Bernoulli factors.  Then \cref{thm:bernoulliFactor-num} rewrites
each factor as the corresponding Bernoulli numerator.  The finite matrix index
is identified with the classical range $1\le j$, $2j\le p-3$.
\end{proof}

\begin{theorem}\label{thm:completed-log-relation}
  \lean{KummerCriterion.CyclotomicUnits.completedLog_relation_of_CPlus_product_mem_powers}\leanok
  \uses{def:CPlusExponentProduct, def:pPowerSubgroup}
If a standard exponent product in $C^+$ is a $p$th power in the full unit
group, then its completed logarithm is divisible by $p$ in the Dwork fixed
subalgebra.
\end{theorem}
\begin{proof}\leanok
The completed logarithm is additive on the product expression and sends a
$p$th power to $p$ times the logarithm of the root.  Expanding the product
therefore expresses the weighted sum of Kummer logarithm columns as a multiple
of $p$.
\end{proof}

\begin{theorem}\label{thm:kummer-mulvec-zero}
  \lean{KummerCriterion.CyclotomicUnits.concreteKummerLogMatrix_mulVec_exponents_eq_zero}\leanok
  \uses{def:concreteKummerLogMatrix, thm:completed-log-relation}
The completed-log divisibility relation gives
\[
  M e=0
\]
over $\mathbb F_p$, where $e$ is the vector of exponents modulo $p$.
\end{theorem}
\begin{proof}\leanok
Take coordinates in the Dwork basis.  Divisibility by $p$ makes every
coordinate vanish after reduction modulo $p$, and those coordinates are
exactly the entries of the matrix-vector product $Me$.
\end{proof}

\begin{theorem}\label{thm:kummer-linear-algebra}
  \lean{KummerCriterion.CyclotomicUnits.exponents_modP_eq_zero_of_kummerLogMatrix_relation}\leanok
  \uses{def:concreteKummerLogMatrix}
If $\det M\ne0$ and $Me=0$, then every coordinate of $e$ is zero modulo $p$.
\end{theorem}
\begin{proof}\leanok
Over the field $\mathbb F_p$, a square matrix with nonzero determinant has
trivial right kernel.
\end{proof}

\begin{theorem}\label{thm:kummer-log-exponents-zero}
  \lean{KummerCriterion.CyclotomicUnits.CPlusGenerator_exponents_modP_zero_of_kummerLog_det_ne_zero}\leanok
  \uses{thm:completed-log-relation, thm:kummer-mulvec-zero,
  thm:kummer-linear-algebra}
If the Kummer logarithm determinant is nonzero, then every exponent product in
$C^+$ which is a $p$th power in the full unit group has all standard exponents
zero modulo $p$.
\end{theorem}
\begin{proof}\leanok
Apply \cref{thm:completed-log-relation} to the $p$th-power relation.  Convert
the logarithmic divisibility statement to $Me=0$ using
\cref{thm:kummer-mulvec-zero}.  Since the determinant is nonzero,
\cref{thm:kummer-linear-algebra} forces $e=0$.
\end{proof}

\begin{theorem}\label{thm:kummer-log-det-saturation}
  \lean{KummerCriterion.CyclotomicUnits.cyclotomicUnits_pSaturated_of_kummerLog_det_ne_zero}\leanok
  \uses{thm:CPlus-pSaturated-exponents, thm:kummer-log-exponents-zero}
If the Kummer logarithm determinant is nonzero, then $C^+$ is $p$-saturated in
the full plus-side unit group.
\end{theorem}
\begin{proof}\leanok
\Cref{thm:kummer-log-exponents-zero} gives exactly the exponent-vanishing
hypothesis required by \cref{thm:CPlus-pSaturated-exponents}.
\end{proof}

\section{The prime-conductor index theorem}

\begin{definition}\label{def:cyclotomicUnitIndexSubgroup}
  \lean{KummerCriterion.cyclotomicUnitIndexSubgroup}\leanok
  \uses{def:normalizedCPlus}
The index-theorem subgroup is the normalized cyclotomic-unit subgroup.
\end{definition}

\begin{theorem}\label{thm:index-subgroup-eq-CPlus}
  \lean{KummerCriterion.cyclotomicUnitIndexSubgroup_eq_CPlus}\leanok
  \uses{def:cyclotomicUnitIndexSubgroup, def:CPlus}
The subgroup used in the index theorem agrees with the subgroup $C^+$ used in
the saturation argument.
\end{theorem}
\begin{proof}\leanok
The proof gives both inclusions.  One direction shows that the normalized
generators lie in the subgroup generated by the standard real cyclotomic
units; the other direction shows that the standard generators are obtained
from the normalized family.
\end{proof}

\begin{theorem}\label{thm:deleted-fourier-kummer-dirichlet}
  \lean{KummerCriterion.kummerDirichletDeterminant_of_deletedFourier}\leanok
The deleted Fourier determinant calculation supplies the
Kummer--Dirichlet determinant required by the Sinnott index pipeline.
\end{theorem}
\begin{proof}\leanok
The proof composes the deleted Fourier determinant identity with the
regulator identity for the cyclotomic-unit family, producing the exact
`KummerDirichletDeterminant` hypothesis needed downstream.
\end{proof}

\begin{theorem}\label{thm:index-pprimary-aux}
  \lean{KummerCriterion.cyclotomicUnitIndex_primeConductor_pPrimary_aux}\leanok
  \uses{thm:index-subgroup-eq-CPlus, thm:deleted-fourier-kummer-dirichlet,
  thm:CPlus-normalized-index}
For $p\ge5$, the normalized cyclotomic-unit index has the same
$p$-primary divisibility as $h^+$:
\[
  p\mid[(\mathcal O_{K^+})^\times:C^+_{\mathrm{norm}}]
  \Longleftrightarrow
  p\mid h^+(K).
\]
\end{theorem}
\begin{proof}\leanok
The deleted Fourier determinant gives the Kummer--Dirichlet determinant.
Sinnott's prime-conductor index theorem applies to the corresponding index
subgroup.  \Cref{thm:index-subgroup-eq-CPlus} identifies that subgroup with
$C^+$, and \cref{thm:CPlus-normalized-index} passes between the standard and
normalized indices at the odd prime $p$.
\end{proof}

\begin{theorem}\label{thm:sinnott-index-pprimary}
  \lean{KummerCriterion.cyclotomicUnitIndex_primeConductor_pPrimary}\leanok
  \uses{thm:index-pprimary-aux}
For every odd prime conductor,
\[
  p\mid[(\mathcal O_{K^+})^\times:C^+_{\mathrm{norm}}]
  \Longleftrightarrow
  p\mid h^+(K).
\]
\end{theorem}
\begin{proof}\leanok
If $p=3$, the normalized subgroup is the whole unit group and $h^+=1$, so
both divisibility statements are false.  If $p\ge5$, the theorem is
\cref{thm:index-pprimary-aux}.
\end{proof}
