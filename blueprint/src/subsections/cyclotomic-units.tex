% Real cyclotomic units, Kummer logarithms, saturation, and the
% prime-conductor index theorem.

This chapter treats the plus side of the criterion.  The real cyclotomic
units generate a finite-index subgroup $C^+$ of the unit group of
$\mathcal O_{K^+}$, and two facts about it are proved.  First, if no
Bernoulli numerator in Kummer's range is divisible by $p$, then $C^+$ is
\emph{$p$-saturated} in the full unit group, hence the index
$[(\mathcal O_{K^+})^\times : C^+]$ is prime to $p$; the tool here is the
Kummer logarithm matrix, whose determinant factors as a diagonal of
Bernoulli numbers times a nonvanishing Vandermonde determinant.  Second,
Sinnott's prime-conductor index theorem \cite{sinnott1978} identifies the
$p$-divisibility of this index with the $p$-divisibility of $h^+$.

\section{The real cyclotomic-unit subgroup}

\begin{definition}\label{def:real-cyclotomic-unit}
  \lean{KummerCriterion.realCyclotomicUnit}\leanok
For $2\le a\le (p-1)/2$, the \emph{real cyclotomic unit} $\varepsilon_a$ is
the unit of $\mathcal O_{K^+}$ obtained by descending the product of the
cyclotomic unit $(1-\zeta_p^a)/(1-\zeta_p)$ of $\mathcal O_K$ with its
complex conjugate:
\[
  \varepsilon_a
  \;=\;
  \frac{(1-\zeta_p^a)\,(1-\zeta_p^{-a})}{(1-\zeta_p)\,(1-\zeta_p^{-1})} .
\]
This element is fixed by complex conjugation, hence lies in
$(\mathcal O_{K^+})^\times$.
\end{definition}

\begin{definition}\label{def:CPlus-generator}
  \lean{KummerCriterion.CPlusGenerator}\leanok
  \uses{def:real-cyclotomic-unit}
The \emph{standard generators} are the $(p-3)/2$ real cyclotomic units
\[
  \varepsilon_2,\ \varepsilon_3,\ \ldots,\ \varepsilon_{(p-1)/2}
\]
of \cref{def:real-cyclotomic-unit}, indexed in the formalisation by
$i \in \{0, \ldots, (p-5)/2\}$ via $a = i + 2$.
\end{definition}

\begin{definition}\label{def:CPlus}
  \lean{KummerCriterion.CPlus}\leanok
  \uses{def:CPlus-generator}
The subgroup $C^+\subseteq(\mathcal O_{K^+})^\times$ is generated by $-1$
and the standard real cyclotomic units:
\[
  C^+=\bigl\langle -1,\ \varepsilon_2,\ldots,\varepsilon_{(p-1)/2}
  \bigr\rangle.
\]
\end{definition}

\begin{definition}\label{def:normalizedCPlus}
  \lean{KummerCriterion.normalizedCPlus}\leanok
  \uses{def:CPlus}
For $2 \le a \le (p-1)/2$, the \emph{normalised cyclotomic unit} is
\[
  \xi_a \;=\; \zeta_p^{\,(1-a)/2}\,\frac{1-\zeta_p^a}{1-\zeta_p},
\]
where the exponent $(1-a)/2$ is read modulo $p$.  The twist by the root of
unity makes $\xi_a$ fixed by complex conjugation, so it defines a unit of
$\mathcal O_{K^+}$.  The normalised subgroup $C^+_{\mathrm{norm}}$ is
generated by $-1$ and the units $\xi_a$.
\end{definition}

\begin{theorem}\label{thm:normalized-square}
  \lean{KummerCriterion.normalizedCPlusGenerator_sq_eq_CPlusGenerator}\leanok
  \uses{def:CPlus, def:normalizedCPlus}
Each normalised generator squares to the corresponding standard generator:
\[
  \xi_a^{\,2} = \varepsilon_a
  \qquad (2 \le a \le (p-1)/2).
\]
\end{theorem}
\begin{proof}\leanok
Using the identity $1-\zeta_p^{-a} = -\zeta_p^{-a}(1-\zeta_p^a)$ in the
numerator and denominator of $\varepsilon_a$,
\[
  \varepsilon_a
  = \frac{(1-\zeta_p^a)\cdot\bigl(-\zeta_p^{-a}\bigr)(1-\zeta_p^a)}
         {(1-\zeta_p)\cdot\bigl(-\zeta_p^{-1}\bigr)(1-\zeta_p)}
  = \zeta_p^{\,1-a}\,\frac{(1-\zeta_p^a)^2}{(1-\zeta_p)^2}
  = \xi_a^{\,2}.
\]
\end{proof}

\begin{theorem}\label{thm:CPlus-normalized-index}
  \lean{KummerCriterion.CPlus_index_prime_dvd_iff_normalizedCPlus_index_prime_dvd}\leanok
  \uses{thm:normalized-square}
For an odd prime $p$,
\[
  p\mid[(\mathcal O_{K^+})^\times:C^+]
  \Longleftrightarrow
  p\mid[(\mathcal O_{K^+})^\times:C^+_{\mathrm{norm}}].
\]
\end{theorem}
\begin{proof}\leanok
By \cref{thm:normalized-square}, $C^+ \subseteq C^+_{\mathrm{norm}}$ and
every square of a normalised generator lies in $C^+$, so the quotient
$C^+_{\mathrm{norm}}/C^+$ is an elementary abelian $2$-group of bounded
rank; in particular the two indices differ by a power of $2$.  Since $p$
is odd, multiplying or dividing by a power of $2$ does not affect
$p$-divisibility.
\end{proof}

\section{\texorpdfstring{Exact $p$-saturation}{Exact p-saturation}}

\begin{definition}\label{def:EPlus}
  \lean{KummerCriterion.EPlus}\leanok
The ambient group for the saturation argument is the full unit group
\[
  E^+ = (\mathcal O_{K^+})^\times .
\]
\end{definition}

\begin{definition}\label{def:pPowerSubgroup}
  \lean{KummerCriterion.pPowerSubgroup}\leanok
For a subgroup $H$ of an abelian group, $H^p$ denotes the exact image of
the $p$th-power map:
\[
  H^p=\{h^p:h\in H\}
\]
(an actual image, not the subgroup generated by it; for abelian groups the
two coincide).
\end{definition}

\begin{definition}\label{def:pSaturated}
  \lean{KummerCriterion.pSaturated}\leanok
  \uses{def:pPowerSubgroup}
A subgroup $H\le E$ is \emph{$p$-saturated} in $E$ when
\[
  H\cap E^p\subseteq H^p,
\]
that is, when every element of $H$ that is a $p$th power in $E$ is already
the $p$th power of an element of $H$.
\end{definition}

\begin{definition}\label{def:CPlusExponentProduct}
  \lean{KummerCriterion.CPlusExponentProduct}\leanok
  \uses{def:CPlus}
An \emph{exponent product} is an expression
\[
  (-1)^s\prod_{a=2}^{(p-1)/2}\varepsilon_a^{\,e_a}
\]
with integer exponents $s, e_a \in \Z$, packaged together with the element
of $C^+$ it represents.
\end{definition}

\begin{theorem}\label{thm:exists-CPlusExponentProduct}
  \lean{KummerCriterion.exists_CPlusExponentProduct_of_mem_CPlus}\leanok
  \uses{def:CPlusExponentProduct}
Every element of $C^+$ admits an exponent-product expression.
\end{theorem}
\begin{proof}\leanok
Induction over membership in the subgroup generated by $-1$ and the
$\varepsilon_a$: each generator trivially has such an expression, the
identity is the empty product, and the set of elements admitting an
expression is closed under multiplication (add the exponent vectors) and
inverses (negate them).
\end{proof}

\begin{theorem}\label{thm:CPlus-pSaturated-exponents}
  \lean{KummerCriterion.CPlus_pSaturated_of_generator_exponents_modP_zero}\leanok
  \uses{def:pSaturated, thm:exists-CPlusExponentProduct}
Assume that whenever an exponent product
\[
  (-1)^s\prod_a\varepsilon_a^{\,e_a}
\]
is a $p$th power in $E^+$, every exponent $e_a$ is divisible by $p$.  Then
$C^+$ is $p$-saturated in $E^+$.
\end{theorem}
\begin{proof}\leanok
Let $x \in C^+$ be a $p$th power in $E^+$, and write $x = (-1)^s \prod_a
\varepsilon_a^{e_a}$ by \cref{thm:exists-CPlusExponentProduct}.  By
hypothesis $p \mid e_a$ for every $a$.  Since $p$ is odd, $(-1)^s =
\bigl((-1)^s\bigr)^p$, so
\[
  x = \Bigl((-1)^s\prod_a\varepsilon_a^{\,e_a/p}\Bigr)^{\!p}
\]
exhibits $x$ as the $p$th power of an element of $C^+$.
\end{proof}

\begin{theorem}\label{thm:not-dvd-index-of-pSaturated}
  \lean{KummerCriterion.not_dvd_index_of_pSaturated}\leanok
  \uses{def:pSaturated, def:CPlus, def:EPlus}
If $C^+$ is $p$-saturated in $E^+$, then
\[
  p\nmid[E^+:C^+].
\]
\end{theorem}
\begin{proof}\leanok
The group-theoretic statement is: a finite-index $p$-saturated subgroup $H
\le E$ whose inclusion captures the $p$-torsion of $E/H$ has index prime to
$p$.  Indeed, if $p$ divided the index, the finite abelian quotient $E/H$
would contain an element of order exactly $p$, say the class of $x$; then
$x^p \in H$ is a $p$th power in $E$, so by saturation $x^p = y^p$ with
$y\in H$, and $x/y$ is a $p$-torsion unit of $E$ not in $H$.  For
$E = E^+$, torsion units of the totally real field $K^+$ are $\pm1$, and
$-1 \in C^+$ by construction, so no such element exists --- a
contradiction.  Finiteness of the index is supplied by the cyclotomic-unit
index theory (the units $\varepsilon_a$ are a full-rank family).
\end{proof}

\section{The Kummer logarithm matrix}

\begin{definition}\label{def:kummerLogRank}
  \lean{KummerCriterion.CyclotomicUnits.kummerLogRank}\leanok
Set
\[
  r \;=\; \frac{p-3}{2},
\]
the common cardinality of the set of standard generators
$\varepsilon_2,\ldots,\varepsilon_{(p-1)/2}$ and of the even index set
$\{2j : 1 \le j,\ 2j \le p-3\}$ of Kummer's range.
\end{definition}

\begin{definition}\label{def:concreteKummerLogMatrix}
  \lean{KummerCriterion.CyclotomicUnits.concreteKummerLogMatrix}\leanok
  \uses{def:kummerLogRank, def:CPlus-generator}
The \emph{Kummer logarithm matrix} $M$ is the $r \times r$ matrix over
$\F_p$ whose $a$th column consists of the coordinates, in the Dwork basis
of the relevant fixed subalgebra, of the completed $p$-adic logarithm of
the generator $\varepsilon_a$, reduced modulo $p$.
\end{definition}

\begin{theorem}\label{thm:kummer-log-vandermonde}
  \lean{KummerCriterion.CyclotomicUnits.concreteKummerLogMatrix_eq_diagonal_mul_vandermonde}\leanok
  \uses{def:concreteKummerLogMatrix}
The Kummer logarithm matrix factors as
\[
  M = D \cdot V,
\]
where $D$ is the diagonal matrix of Bernoulli factors (the $j$th diagonal
entry being a unit multiple of the reduction of $B_{2j}/2j$) and $V$ is a
Vandermonde-type matrix built from Teichm\"uller values.
\end{theorem}
\begin{proof}\leanok
Expand the completed logarithm of each generator in the Dwork basis and
compute the coefficients.  Each entry of $M$ splits as a product of a
factor depending only on the row $j$ --- the Bernoulli factor --- and a
factor of the form $\eta_a^{\,2j}$ depending on the column through a
Teichm\"uller node $\eta_a$.  Collecting the row factors into a diagonal
matrix exhibits the remaining matrix as a Vandermonde matrix in the nodes
$\eta_a^2$.
\end{proof}

\begin{theorem}\label{thm:vandermonde-nonzero}
  \lean{KummerCriterion.CyclotomicUnits.vandermonde_teichmuller_even_sub_one_det_ne_zero}\leanok
The Teichm\"uller Vandermonde determinant occurring in
\cref{thm:kummer-log-vandermonde} is nonzero.
\end{theorem}
\begin{proof}\leanok
The nodes are the values $\eta_a^2$ for $2 \le a \le (p-1)/2$; they are
pairwise distinct and distinct from $1$, because the Teichm\"uller
character is injective on $(\Z/p\Z)^\times$ and squaring is two-to-one with
the representatives $a$ chosen in a half-system.  A Vandermonde
determinant in distinct nodes is a product of nonzero differences, hence
nonzero.
\end{proof}

\begin{theorem}\label{thm:kummer-det-bernoulli-factor}
  \lean{KummerCriterion.CyclotomicUnits.concreteKummerLogMatrix_det_ne_zero_iff_forall_bernoulliFactor_ne_zero}\leanok
  \uses{thm:kummer-log-vandermonde, thm:vandermonde-nonzero}
$\det M \ne 0$ if and only if every diagonal Bernoulli factor is nonzero.
\end{theorem}
\begin{proof}\leanok
By \cref{thm:kummer-log-vandermonde}, $\det M = \det D \cdot \det V$, and
$\det V \ne 0$ by \cref{thm:vandermonde-nonzero}.  Hence $\det M \ne 0$
exactly when $\det D = \prod_j D_{jj} \ne 0$, i.e.\ when every diagonal
entry is nonzero.
\end{proof}

\begin{theorem}\label{thm:bernoulliFactor-num}
  \lean{KummerCriterion.CyclotomicUnits.bernoulliFactor_ne_zero_iff_not_dvd_bernoulli_num}\leanok
  \uses{lem:padic-bernoulli-factor}
The diagonal Bernoulli factor attached to $j$ is nonzero in $\F_p$ exactly
when
\[
  p\nmid\operatorname{num}(B_{2j}).
\]
\end{theorem}
\begin{proof}\leanok
The factor is the reduction modulo $p$ of a $p$-integral rational
expression which is a $p$-adic unit multiple of $B_{2j}/2j$; in the range
$2j \le p-3$, both $2j$ and the denominator of $B_{2j}$ are prime to $p$
(\cref{lem:padic-bernoulli-factor}).  So the reduction vanishes exactly
when $p$ divides the numerator of $B_{2j}$.
\end{proof}

\begin{theorem}\label{thm:kummer-log-det-bernoulli}
  \lean{KummerCriterion.CyclotomicUnits.kummerLogMatrix_det_ne_zero_iff_bernoulli_nonzero}\leanok
  \uses{thm:kummer-det-bernoulli-factor, thm:bernoulliFactor-num}
The Kummer logarithm determinant is nonzero precisely when no Bernoulli
numerator in Kummer's range is divisible by $p$:
\[
  \det M\ne0
  \Longleftrightarrow
  \forall j,\ 1\le j,\ 2j\le p-3,\quad
  p\nmid\operatorname{num}(B_{2j}).
\]
\end{theorem}
\begin{proof}\leanok
\Cref{thm:kummer-det-bernoulli-factor} reduces the determinant condition to
the nonvanishing of every diagonal Bernoulli factor, and
\cref{thm:bernoulliFactor-num} translates each of those conditions into
nondivisibility of the corresponding numerator.  The finite index set of
the matrix is identified with the classical range $1\le j$, $2j\le p-3$.
\end{proof}

\begin{theorem}\label{thm:completed-log-relation}
  \lean{KummerCriterion.CyclotomicUnits.completedLog_relation_of_CPlus_product_mem_powers}\leanok
  \uses{def:CPlusExponentProduct, def:pPowerSubgroup}
If an exponent product in $C^+$ is a $p$th power in the full unit group,
then its completed logarithm is divisible by $p$ in the Dwork fixed
subalgebra.
\end{theorem}
\begin{proof}\leanok
The completed logarithm is a homomorphism on the relevant units: it turns
the exponent product into the corresponding integer combination of the
logarithms of the generators, and it sends a $p$th power $y^p$ to
$p\log y$.  Equating the two expressions exhibits the combination
$\sum_a e_a \log \varepsilon_a$ as $p$ times an element of the fixed
subalgebra.  (The sign $(-1)^s$ is killed by the logarithm.)
\end{proof}

\begin{theorem}\label{thm:kummer-mulvec-zero}
  \lean{KummerCriterion.CyclotomicUnits.concreteKummerLogMatrix_mulVec_exponents_eq_zero}\leanok
  \uses{def:concreteKummerLogMatrix, thm:completed-log-relation}
In the situation of \cref{thm:completed-log-relation},
\[
  M e=0
  \quad\text{over } \F_p,
\]
where $e$ is the vector of exponents $e_a$ reduced modulo $p$.
\end{theorem}
\begin{proof}\leanok
Take coordinates in the Dwork basis.  The $j$th coordinate of the
combination $\sum_a e_a \log\varepsilon_a$ is the $j$th entry of the
matrix-vector product $Me$ (before reduction).  Divisibility by $p$ of the
combination makes every coordinate divisible by $p$, i.e.\ every entry of
$Me$ vanishes after reduction modulo $p$.
\end{proof}

\begin{theorem}\label{thm:kummer-linear-algebra}
  \lean{KummerCriterion.CyclotomicUnits.exponents_modP_eq_zero_of_kummerLogMatrix_relation}\leanok
  \uses{def:concreteKummerLogMatrix}
If $\det M\ne0$ and $Me=0$, then $e = 0$ in $\F_p^{\,r}$.
\end{theorem}
\begin{proof}\leanok
Over the field $\F_p$, a square matrix with nonzero determinant is
invertible, so its kernel is trivial.
\end{proof}

\begin{theorem}\label{thm:kummer-log-exponents-zero}
  \lean{KummerCriterion.CyclotomicUnits.CPlusGenerator_exponents_modP_zero_of_kummerLog_det_ne_zero}\leanok
  \uses{thm:completed-log-relation, thm:kummer-mulvec-zero,
  thm:kummer-linear-algebra}
If $\det M \ne 0$, then every exponent product in $C^+$ which is a $p$th
power in the full unit group has all exponents $e_a$ divisible by $p$.
\end{theorem}
\begin{proof}\leanok
Apply \cref{thm:completed-log-relation} to the $p$th-power relation, then
\cref{thm:kummer-mulvec-zero} converts the logarithmic divisibility into
$Me=0$, and \cref{thm:kummer-linear-algebra} forces $e=0$ modulo $p$.
\end{proof}

\begin{theorem}\label{thm:kummer-log-det-saturation}
  \lean{KummerCriterion.CyclotomicUnits.cyclotomicUnits_pSaturated_of_kummerLog_det_ne_zero}\leanok
  \uses{thm:CPlus-pSaturated-exponents, thm:kummer-log-exponents-zero}
If $\det M \ne 0$, then $C^+$ is $p$-saturated in
$(\mathcal O_{K^+})^\times$.
\end{theorem}
\begin{proof}\leanok
\Cref{thm:kummer-log-exponents-zero} is exactly the exponent-vanishing
hypothesis of the saturation criterion
\cref{thm:CPlus-pSaturated-exponents}.
\end{proof}

\section{The prime-conductor index theorem}

\begin{definition}\label{def:cyclotomicUnitIndexSubgroup}
  \lean{KummerCriterion.cyclotomicUnitIndexSubgroup}\leanok
  \uses{def:normalizedCPlus}
The subgroup featuring in the prime-conductor index theorem is the
normalised cyclotomic-unit subgroup $C^+_{\mathrm{norm}}$ of
\cref{def:normalizedCPlus}.
\end{definition}

\begin{theorem}\label{thm:index-subgroup-eq-CPlus}
  \lean{KummerCriterion.cyclotomicUnitIndexSubgroup_eq_CPlus}\leanok
  \uses{def:cyclotomicUnitIndexSubgroup, def:CPlus}
Up to the normalisation of \cref{thm:CPlus-normalized-index}, the subgroup
used in the index theorem agrees with the subgroup $C^+$ used in the
saturation argument: each contains representatives generating the other
modulo $\{\pm1\}$ and squares.
\end{theorem}
\begin{proof}\leanok
Both inclusions are checked on generators.  A normalised generator $\xi_a$
times a root of unity recovers the cyclotomic unit
$(1-\zeta_p^a)/(1-\zeta_p)$, whose product with its conjugate is
$\varepsilon_a = \xi_a^2$; conversely each $\varepsilon_a$ is the square of
$\xi_a$, so the two subgroups generate each other up to squares and signs.
\end{proof}

\begin{theorem}\label{thm:deleted-fourier-kummer-dirichlet}
  \lean{KummerCriterion.kummerDirichletDeterminant_of_deletedFourier}\leanok
The deleted-Fourier determinant identity supplies the Kummer--Dirichlet
determinant hypothesis required by the Sinnott index pipeline: the
regulator-type determinant of the cyclotomic-unit family is, up to an
explicit nonzero factor, a product of the character sums
$\sum_a \chi(a) \log|1-e^{2\pi i a/p}|$ over the even nontrivial
characters.
\end{theorem}
\begin{proof}\leanok
The logarithmic embeddings of the cyclotomic units form a matrix whose
group-theoretic structure (rows and columns indexed by a cyclic group) is
diagonalised by the discrete Fourier transform with one character deleted.
Composing the resulting determinant identity with the regulator identity
for the cyclotomic-unit family produces exactly the Kummer--Dirichlet
determinant needed downstream.
\end{proof}

\begin{theorem}\label{thm:index-pprimary-aux}
  \lean{KummerCriterion.cyclotomicUnitIndex_primeConductor_pPrimary_aux}\leanok
  \uses{thm:index-subgroup-eq-CPlus, thm:deleted-fourier-kummer-dirichlet,
  thm:CPlus-normalized-index, def:class-number-hplus}
For $p\ge5$, the normalised cyclotomic-unit index has the same
$p$-divisibility as $h^+$:
\[
  p\mid[(\mathcal O_{K^+})^\times:C^+_{\mathrm{norm}}]
  \Longleftrightarrow
  p\mid h^+(K).
\]
\end{theorem}
\begin{proof}\leanok
This is Sinnott's prime-conductor index theorem: for prime conductor the
index of the circular units in the full unit group equals $h^+$.  The
deleted-Fourier determinant of
\cref{thm:deleted-fourier-kummer-dirichlet} provides the determinant
computation at the heart of the proof,
\cref{thm:index-subgroup-eq-CPlus} identifies the index subgroup with the
one used here, and \cref{thm:CPlus-normalized-index} transfers
$p$-divisibility between the standard and normalised indices.
\end{proof}

\begin{theorem}\label{thm:sinnott-index-pprimary}
  \lean{KummerCriterion.cyclotomicUnitIndex_primeConductor_pPrimary}\leanok
  \uses{thm:index-pprimary-aux}
For every odd prime conductor,
\[
  p\mid[(\mathcal O_{K^+})^\times:C^+_{\mathrm{norm}}]
  \Longleftrightarrow
  p\mid h^+(K).
\]
\end{theorem}
\begin{proof}\leanok
For $p=3$ the field $K^+$ is $\Q$: the normalised subgroup is the whole
unit group $\{\pm1\}$ and $h^+=1$, so both sides are false.  For $p\ge5$
this is \cref{thm:index-pprimary-aux}.
\end{proof}
