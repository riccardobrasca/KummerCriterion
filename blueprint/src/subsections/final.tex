% Final assembly of Kummer's criterion.

This chapter assembles the preceding ingredients into Kummer's criterion.  The
mathematical content is the equivalence
\[
  p\text{ regular}
  \quad\Longleftrightarrow\quad
  \forall k,\ 1\le k,\ 2k\le p-3,\quad
  p\nmid (B_{2k})_{\mathrm{num}}.
\]

\section{The minus class-number criterion}

The analytic and $p$-adic work on the relative class number gives the
following criterion.

\begin{theorem}[Minus class-number criterion]
  \label{thm:hminus-bernoulli-final}
  \lean{KummerCriterion.p_dvd_hMinus_iff_p_dvd_some_bernoulli}
  \leanok
  \uses{thm:hminus-formula, thm:boundary-character-bernoulli,
    thm:bernoulli-padicInt-below, cor:bernoulli-den-coprime}
Let $p$ be an odd prime and let $K$ be a cyclotomic field of conductor $p$.
Then
\[
  p\mid h^-(K)
  \quad\Longleftrightarrow\quad
  \exists k,\ 1\le k,\ 2k\le p-3,\quad
  p\mid (B_{2k})_{\mathrm{num}}.
\]
\end{theorem}

\begin{proof}\leanok
Starting from the analytic class-number formula for $h^-$, one rewrites the
odd-character product in terms of Teichmuller characters.  The boundary
character contributes the unique factor containing the pole at $p-1$; after
the explicit leading factor $2p$ is included, that boundary contribution is
congruent to $1$ modulo $p$.  The remaining factors are
\[
  -\frac12 B_{1,\omega^j}
  \qquad (1\le j\le p-4,\ j\text{ odd}).
\]
Kummer congruences replace these factors modulo $p$ by
\[
  -\frac12\,\frac{B_{j+1}}{j+1}.
\]
Writing $j+1=2k$, the range becomes exactly
$1\le k$ and $2k\le p-3$.

For these indices the denominator of $B_{2k}$ is prime to $p$, and so is
$2k$.  Hence the corresponding $p$-adic factor is a unit if and only if
$p$ does not divide the numerator of $B_{2k}$.  A finite product in
$\mathbb Z_p$ is a non-unit if and only if one of its factors is a non-unit.
This identifies $p\mid h^-(K)$ with the existence of a Bernoulli numerator in
the displayed range divisible by $p$.
\end{proof}

We shall also use the contrapositive form of this criterion.

\begin{corollary}[Non-divisibility of $h^-$ gives Bernoulli non-divisibility]
  \label{cor:hminus-not-dvd-bernoulli-nonzero}
  \lean{KummerCriterion.bernoulli_nonzero_of_not_dvd_hMinus}
  \leanok
  \uses{thm:hminus-bernoulli-final}
If $p\nmid h^-(K)$, then for every $j$ with $1\le j$ and $2j\le p-3$,
\[
  p\nmid (B_{2j})_{\mathrm{num}}.
\]
\end{corollary}

\begin{proof}\leanok
This is the contrapositive of \cref{thm:hminus-bernoulli-final}.  If some
Bernoulli numerator in the range were divisible by $p$, the right-hand side
of the criterion would hold, and therefore $p\mid h^-(K)$.
\end{proof}

\section{The plus-to-minus divisibility step}

The cyclotomic-unit theorem from \cref{ch:cyclotomic-units} turns the
contrapositive of the minus criterion into a statement about $h^+$.

\begin{theorem}[Contrapositive plus-to-minus divisibility]
  \label{thm:not-dvd-hplus-of-not-dvd-hminus-units}
  \lean{KummerCriterion.not_dvd_hPlus_of_not_dvd_hMinus_units}
  \leanok
  \uses{cor:hminus-not-dvd-bernoulli-nonzero, thm:bernoulli-nonzero-index,
    thm:cyclotomic-unit-index-hplus, lem:CPlus-index-normalized-index}
Let $p$ be an odd prime and let $K$ be a cyclotomic field of conductor $p$.
If
\[
  p\nmid h^-(K),
\]
then
\[
  p\nmid h^+(K).
\]
\end{theorem}

\begin{proof}\leanok
Assume $p\nmid h^-(K)$.  By
\cref{cor:hminus-not-dvd-bernoulli-nonzero}, all Bernoulli numerators
$(B_{2j})_{\mathrm{num}}$ in Kummer's range are prime to $p$.  The
cyclotomic-unit theorem
\cref{thm:bernoulli-nonzero-index} therefore gives
\[
  p\nmid [E^+:C^+_{\mathrm{sq}}].
\]

Suppose, for contradiction, that $p\mid h^+(K)$.  The prime-conductor
cyclotomic-unit index theorem
\cref{thm:cyclotomic-unit-index-hplus} gives
\[
  p\mid [E^+:C^+_{\mathrm{norm}}].
\]
By the odd-primary comparison of the two cyclotomic-unit subgroups,
\cref{lem:CPlus-index-normalized-index}, this is equivalent to
\[
  p\mid [E^+:C^+_{\mathrm{sq}}],
\]
contradicting the index non-divisibility already obtained.  Hence
$p\nmid h^+(K)$.
\end{proof}

\begin{corollary}[Plus divisibility implies minus divisibility]
  \label{cor:hplus-to-hminus-units}
  \lean{KummerCriterion.weakReflection_dvd_hMinus_of_dvd_hPlus_units}
  \leanok
  \uses{thm:not-dvd-hplus-of-not-dvd-hminus-units}
If
\[
  p\mid h^+(K),
\]
then
\[
  p\mid h^-(K).
\]
\end{corollary}

\begin{proof}\leanok
Argue by contradiction.  If $p\nmid h^-(K)$, then
\cref{thm:not-dvd-hplus-of-not-dvd-hminus-units} gives
$p\nmid h^+(K)$, contradicting the hypothesis.  Therefore
$p\mid h^-(K)$.
\end{proof}

\section{The total class number}

Recall from \cref{thm:h-factorisation} that
\[
  h(K)=h^+(K)\,h^-(K).
\]
The previous corollary lets us reduce divisibility of the total class number
to divisibility of the minus class number.

\begin{theorem}[Total class-number divisibility]
  \label{thm:dvd-h-iff-dvd-hminus-units}
  \lean{KummerCriterion.dvd_h_iff_dvd_hMinus_of_dvd_hPlus_imp}
  \leanok
  \uses{thm:h-factorisation, cor:hplus-to-hminus-units}
For odd $p$, assume that
\[
  p\mid h^+(K)\Longrightarrow p\mid h^-(K).
\]
Then
\[
  p\mid h(K)
  \quad\Longleftrightarrow\quad
  p\mid h^-(K).
\]
\end{theorem}

\begin{proof}\leanok
The implication from right to left is immediate from
$h(K)=h^+(K)h^-(K)$.

Conversely, suppose $p\mid h(K)$.  Since
$h(K)=h^+(K)h^-(K)$ and $p$ is prime, Euclid's lemma gives
\[
  p\mid h^+(K)
  \quad\text{or}\quad
  p\mid h^-(K).
\]
In the second case we are done.  In the first case,
the assumed implication converts $p\mid h^+(K)$ into $p\mid h^-(K)$.
Thus $p\mid h^-(K)$ in all cases.
\end{proof}

\begin{theorem}[Total class number and Bernoulli numerators]
  \label{thm:dvd-h-iff-bernoulli-units}
  \lean{KummerCriterion.dvd_h_iff_exists_dvd_bernoulli_units}
  \leanok
  \uses{thm:dvd-h-iff-dvd-hminus-units, thm:hminus-bernoulli-final,
    cor:hplus-to-hminus-units}
For odd $p$,
\[
  p\mid h(K)
  \quad\Longleftrightarrow\quad
  \exists k,\ 1\le k,\ 2k\le p-3,\quad
  p\mid (B_{2k})_{\mathrm{num}}.
\]
\end{theorem}

\begin{proof}\leanok
Apply \cref{thm:dvd-h-iff-dvd-hminus-units} with the implication supplied by
\cref{cor:hplus-to-hminus-units}.  This replaces $p\mid h(K)$ by
$p\mid h^-(K)$.  Then apply the minus class-number criterion
\cref{thm:hminus-bernoulli-final}, which replaces $p\mid h^-(K)$ by the
existence of a divisible Bernoulli numerator in Kummer's range.
\end{proof}

\section{The final theorem}

\begin{theorem}[Kummer's criterion]
  \label{thm:kummer-final}
  \lean{KummerCriterion}
  \leanok
  \uses{thm:dvd-h-iff-bernoulli-units}
Let $p$ be an odd prime. Then
\[
  \mathrm{IsRegularPrime}(p)
  \quad\Longleftrightarrow\quad
  \forall k,\ 1\le k,\ 2k\le p-3,\quad
  p\nmid (B_{2k})_{\mathrm{num}}.
\]
\end{theorem}

\begin{proof}\leanok
Take $K$ to be the standard cyclotomic field
\[
  K=\Q(\zeta_p).
\]
By definition, regularity of \(p\) is coprimality with the class number of
this standard cyclotomic field.  Since \(p\) is prime, this is equivalent to
\[
  \neg\, p\mid h(\Q(\zeta_p)).
\]
Applying \cref{thm:dvd-h-iff-bernoulli-units} to this \(K\) identifies
divisibility by \(p\) with the existence of a Bernoulli numerator in Kummer's
range divisible by \(p\).  Negating
\[
  \exists k,\ 1\le k,\ 2k\le p-3,\quad p\mid (B_{2k})_{\mathrm{num}}
\]
gives exactly the displayed universal non-divisibility condition.
\end{proof}
