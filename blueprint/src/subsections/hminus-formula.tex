% The analytic h^- formula and its reduction to Bernoulli numerators.

This chapter proves the analytic formula for the relative class number,
\[
  h^- = 2p \prod_{\chi\ \mathrm{odd}} \Bigl(-\tfrac12 B_{1,\chi^{-1}}\Bigr),
\]
and reduces it modulo $p$ to the Bernoulli-numerator criterion for
$p \mid h^-$.  The reduction proceeds in two steps: the odd characters
modulo $p$ are indexed by odd powers of the Teichm\"uller character, and
each generalised Bernoulli value $B_{1,\omega^j}$ is replaced by the
congruent classical quotient $B_{j+1}/(j+1)$.

\section{Characters and Bernoulli factors}

\begin{definition}\label{def:dirichlet-character}
  \lean{DirichletCharacter}\leanok
A \emph{Dirichlet character} modulo $N$ with values in a commutative ring
$R$ is a multiplicative character of $(\Z/N\Z)^\times$ with values in
$R^\times$, extended by zero to all of $\Z/N\Z$.  A character $\chi$
modulo $N > 1$ is \emph{even} if $\chi(-1) = 1$ and \emph{odd} if
$\chi(-1) = -1$.  In this chapter the modulus is the odd prime $p$, in
which case every nontrivial character is primitive and exactly
$(p-1)/2$ of the characters are odd.
\end{definition}

\begin{definition}\label{def:generalized-bernoulli}
  \lean{KummerCriterion.BernoulliGen}\leanok
  \uses{def:dirichlet-character}
For a Dirichlet character $\chi$ modulo $N$ and $n \ge 0$, the
\emph{generalised Bernoulli number} is
\[
  B_{n,\chi}
  \;=\;
  N^{\,n-1} \sum_{a \bmod N} \chi(a)\, B_n\!\left(\frac{a}{N}\right),
\]
where $B_n(X)$ is the $n$th Bernoulli polynomial.  For nontrivial $\chi$
the case $n = 1$ simplifies to
\[
  B_{1,\chi} = \frac{1}{N}\sum_{a=1}^{N} \chi(a)\,a ,
\]
because $\sum_a \chi(a) = 0$ kills the constant term of
$B_1(X) = X - \tfrac12$.  The class-number formula only uses the values
$B_{1,\chi^{-1}}$ for odd $\chi$.
\end{definition}

\begin{definition}\label{def:teichmuller}
  \lean{KummerCriterion.teichmullerCharQp}\leanok
  \uses{def:dirichlet-character}
The \emph{Teichm\"uller character} of the prime $p$ is the unique character
\[
  \omega\colon\left(\Z/p\Z\right)^\times\to\Z_p^\times
\]
such that $\omega(a) \equiv a \pmod p$ and $\omega(a)^{p-1} = 1$ for every
$a$; its values are the $(p-1)$st roots of unity in $\Z_p$.  Composing with
the inclusion $\Z_p \hookrightarrow \Q_p$ gives the $\Q_p$-valued Dirichlet
character used below.  The characters $\omega^j$ for $0 \le j \le p-2$
exhaust the characters modulo $p$, and $\omega^j$ is odd exactly when $j$
is odd.
\end{definition}

\begin{theorem}\label{thm:bernoulli-padicInt-below}
  \lean{KummerCriterion.bernoulli_mem_padicInt_of_lt_sub_one}\leanok
For $0<n<p-1$, the classical Bernoulli number $B_n$ is $p$-adically
integral: $B_n \in \Z_p$.
\end{theorem}
\begin{proof}\leanok
By the von Staudt--Clausen theorem, the denominator of $B_n$ (for $n$ even,
the only nontrivial case) is the product of the primes $q$ with
$(q-1)\mid n$.  If $p$ occurred, then $p - 1 \le n$, contradicting
$n < p-1$.  Hence the denominator is prime to $p$ and $B_n \in \Z_p$.
\end{proof}

\begin{theorem}\label{thm:bernoulli-den-coprime}
  \lean{KummerCriterion.prime_not_dvd_bernoulli_den_of_lt_sub_one}\leanok
  \uses{thm:bernoulli-padicInt-below}
For $0<n<p-1$, the prime $p$ does not divide the denominator of $B_n$.
\end{theorem}
\begin{proof}\leanok
This is the rational form of \cref{thm:bernoulli-padicInt-below}: a
rational number lies in $\Z_p$ exactly when its reduced denominator is
prime to $p$.
\end{proof}

\begin{lemma}\label{lem:padic-bernoulli-factor}
  \lean{KummerCriterion.exists_padicInt_bernoulli_factor}\leanok
  \uses{thm:bernoulli-den-coprime}
For $0<n<p-1$, the quotient $B_n/n$ lies in $\Z_p$, and it is a $p$-adic
unit if and only if $p$ does not divide the numerator of $B_n$.
\end{lemma}
\begin{proof}\leanok
The denominator of $B_n$ is prime to $p$ by
\cref{thm:bernoulli-den-coprime}, and $n < p-1 < p$ is also prime to $p$,
so $B_n/n \in \Z_p$.  In $\Z_p$ the non-units are exactly the elements of
$p\Z_p$; since the denominator and $n$ are $p$-adic units, $B_n/n \in
p\Z_p$ holds precisely when $p$ divides the numerator of $B_n$.
\end{proof}

\begin{theorem}\label{thm:teichmuller-bernoulli-congruence}
  \lean{KummerCriterion.bernoulliGen_teichmuller_pow_sModEq_div}\leanok
  \uses{def:generalized-bernoulli, def:teichmuller}
For the odd exponents $j$ occurring in the relative class-number formula
(odd $j$ with $j + 1$ divisible by neither $p$ nor $p-1$),
\[
  B_{1,\omega^j}
  \equiv \frac{B_{j+1}}{j+1}\pmod p .
\]
\end{theorem}
\begin{proof}\leanok
This is the standard congruence linking generalised Bernoulli values of
Teichm\"uller powers to classical Bernoulli numbers.  Writing
$B_{1,\omega^j} = \tfrac1p \sum_a \omega^j(a)\,a$ and using
$\omega(a) \equiv a^{p^r}$ to high $p$-adic accuracy, the sum is compared
with the power sum $\sum_a a^{\,j+1}$, which Faulhaber's formula relates to
$B_{j+1}$; the hypothesis that $j+1$ is divisible by neither $p$ nor $p-1$
makes the comparison factor a $p$-adic unit and keeps the error terms in
$p\Z_p$.
\end{proof}

\section{The relative class-number formula}

\begin{theorem}\label{thm:hplus-even-lvalues}
  \lean{KummerCriterion.hPlus_formula_of_evenLValues}\leanok
  \uses{def:dirichlet-character, def:class-number-hplus}
The plus class number is given by the even part of the cyclotomic
$L$-value product:
\[
  h^+(K)=
  c_{K^+}
  \prod_{\substack{\chi\ \mathrm{even}\\ \chi\ne1}}
  \left(
    -\,\tau(\chi^{-1})^{-1}
    \sum_{a \bmod p} \chi^{-1}(a)\,
      \log\bigl|1-e^{2\pi i a/p}\bigr|
  \right),
\]
where each factor of the product equals $L(1,\chi)$ and $c_{K^+}$ is the
explicit constant of the analytic class-number formula of $K^+$ (built
from the discriminant, the regulator and the roots of unity of $K^+$).
\end{theorem}
\begin{proof}\leanok
The analytic class-number formula for the totally real field $K^+$
expresses $h^+$ through the residue of the Dedekind zeta function
$\zeta_{K^+}$.  Factoring $\zeta_{K^+}(s) = \zeta(s)\prod_{\chi \text{
even},\ \chi\ne1} L(s,\chi)$ over the even characters and inserting the
classical evaluation of $L(1,\chi)$ for even primitive $\chi$ --- the
displayed Gauss-sum-and-logarithm expression --- gives the formula.
\end{proof}

\begin{theorem}\label{thm:raw-gauss-product}
  \lean{KummerCriterion.rawGaussProduct}\leanok
  \uses{def:dirichlet-character}
The product of the Gauss sums over the odd characters modulo $p$ is
\[
  \prod_{\chi\ \mathrm{odd}} \tau(\chi)
  \;=\;
  i^{\,(p-1)/2}\,\sqrt{p^{\,(p-1)/2}} .
\]
\end{theorem}
\begin{proof}\leanok
The proof splits into the cases $p\equiv1\pmod4$ and $p\equiv3\pmod4$.  A
non-quadratic odd character $\chi$ is paired with its distinct inverse
$\chi^{-1}$ (also odd), and the identity
$\tau(\chi)\tau(\chi^{-1}) = \chi(-1)\,p = -p$ evaluates each pair.  When
$p \equiv 3 \pmod 4$ the quadratic character is odd and unpaired; it
contributes the classical quadratic Gauss sum $i\sqrt p$.  Counting pairs
in the two cases and collecting signs and powers of $i$ gives the stated
value.
\end{proof}

\begin{theorem}\label{thm:cyclotomic-gauss-goal}
  \lean{KummerCriterion.cyclotomicHGaussGoal_holds}\leanok
  \uses{thm:raw-gauss-product}
The explicit Gauss-product evaluation of \cref{thm:raw-gauss-product}
satisfies the packaged Gauss hypothesis required by the relative
class-number assembly theorem (\cref{thm:hminus-formula-from-inputs}).
\end{theorem}
\begin{proof}\leanok
Purely formal: the assembly theorem consumes the Gauss product in a
normalised shape (with the analytic constants of the odd $L$-value formula
distributed over the product), and this normalisation is a finite algebraic
rewriting of \cref{thm:raw-gauss-product}.
\end{proof}

\begin{theorem}\label{thm:hminus-formula-from-inputs}
  \lean{KummerCriterion.hMinus_formula_of_residue_and_hPlus_cyclotomic_and_gauss}\leanok
  \uses{def:class-number-hminus, def:generalized-bernoulli,
  thm:hplus-even-lvalues, thm:cyclotomic-gauss-goal}
Assume the factorisation of the Dedekind zeta residue of $K$ into the even
and odd $L$-value products, the $h^+$ formula of
\cref{thm:hplus-even-lvalues}, and the packaged Gauss-product identity.
Then
\[
  h^-(K)
  =
  2p\prod_{\chi\ \mathrm{odd}}
    \left(-\frac12 B_{1,\chi^{-1}}\right).
\]
\end{theorem}
\begin{proof}\leanok
Divide the analytic class-number formula of $K$ (in residue form) by that
of $K^+$.  Using $h = h^+ h^-$, the quotient expresses $h^-$ times the
ratio of the analytic constants as the product of the odd $L$-values
$L(1,\chi)$.  Substituting the odd $L$-value formula
\[
  L(1,\chi) = \frac{\pi i\,\tau(\chi)}{p}\,B_{1,\chi^{-1}}
\]
for each odd $\chi$ and cancelling the powers of $\pi$, $i$ and the Gauss
sums against the analytic constants --- this is exactly what the packaged
Gauss-product identity provides --- leaves the displayed product of
generalised Bernoulli factors with the prefactor $2p$.
\end{proof}

\begin{theorem}\label{thm:hminus-formula}
  \lean{KummerCriterion.hMinus_formula}\leanok
  \uses{thm:hminus-formula-from-inputs, thm:hplus-even-lvalues,
  thm:cyclotomic-gauss-goal}
For an odd prime cyclotomic field,
\[
  h^-(K)
  =
  2p\prod_{\chi\ \mathrm{odd}}
    \left(-\frac12 B_{1,\chi^{-1}}\right).
\]
\end{theorem}
\begin{proof}\leanok
Supply the three inputs of \cref{thm:hminus-formula-from-inputs}: the
residue of $\zeta_K$ factors over the nontrivial characters into the even
and odd $L$-value products; the even product is converted by
\cref{thm:hplus-even-lvalues}; and the Gauss hypothesis is
\cref{thm:cyclotomic-gauss-goal}.
\end{proof}

\section{\texorpdfstring{From $h^-$ to Bernoulli numerators}{From h-minus to Bernoulli numerators}}

\begin{theorem}\label{thm:hminus-teichmuller-mod-p}
  \lean{KummerCriterion.hMinus_formula_teichmuller_mod_p}\leanok
  \uses{thm:hminus-formula, def:teichmuller}
There exists $z\in\Z_p$ such that, inside $\Q_p$,
\[
  h^-(K)
  =
  \prod_{\substack{j<p-2\\ j\ \mathrm{odd}}}
    \left(-\frac12 B_{1,\omega^j}\right)
  + pz .
\]
\end{theorem}
\begin{proof}\leanok
Starting from \cref{thm:hminus-formula}, index the odd characters modulo
$p$ as $\omega^j$ with $j$ odd, $1 \le j \le p-2$.  The boundary exponent
$j = p-2$ plays a special role: the combination of the prefactor $2p$ with
the boundary factor $-\tfrac12 B_{1,\omega^{p-2}}$ is a $p$-adic integer
congruent to $1$ modulo $p$, so it can be absorbed into the error term
$pz$, leaving the product over odd $j < p-2$.
\end{proof}

\begin{theorem}\label{thm:hminus-bernoulli-mod-p}
  \lean{KummerCriterion.hMinus_formula_bernoulli_mod_p}\leanok
  \uses{thm:hminus-teichmuller-mod-p, thm:teichmuller-bernoulli-congruence,
  lem:padic-bernoulli-factor}
There exists $z\in\Z_p$ such that, inside $\Q_p$,
\[
  h^-(K)
  =
  \prod_{\substack{j<p-2\\ j\ \mathrm{odd}}}
    \left(-\frac12\,\frac{B_{j+1}}{j+1}\right)
  + pz .
\]
\end{theorem}
\begin{proof}\leanok
Replace each factor $B_{1,\omega^j}$ in \cref{thm:hminus-teichmuller-mod-p}
by the congruent classical quotient $B_{j+1}/(j+1)$ from
\cref{thm:teichmuller-bernoulli-congruence}; for odd $j < p-2$ the index
$j+1$ is even with $0 < j+1 < p-1$, so the hypotheses of the congruence
hold.  Each replacement changes the product by an element of $p\Z_p$ times
a product of $p$-adic integers --- integrality of the remaining factors is
\cref{lem:padic-bernoulli-factor} --- so all errors collect into a single
multiple of $p$.
\end{proof}

\begin{lemma}\label{lem:odd-index-to-k}
  \lean{KummerCriterion.exists_odd_index_iff_exists_k}\leanok
For any predicate $Q$ on natural numbers,
\[
  \bigl(\exists j,\ j<p-2,\ j\ \mathrm{odd},\ Q(j+1)\bigr)
  \quad\Longleftrightarrow\quad
  \bigl(\exists k,\ 1\le k,\ 2k\le p-3,\ Q(2k)\bigr).
\]
\end{lemma}
\begin{proof}\leanok
If $j$ is odd, write $j=2k-1$ with $k \ge 1$, so $j+1=2k$, and $j < p-2$
becomes $2k \le p-3$ after parity bookkeeping.  Conversely, an even index
$2k$ in Kummer's range corresponds to the odd index $j=2k-1 < p-2$.
\end{proof}

\begin{theorem}\label{thm:hminus-bernoulli-final}
  \lean{KummerCriterion.p_dvd_hMinus_iff_p_dvd_some_bernoulli}\leanok
  \uses{thm:hminus-bernoulli-mod-p, lem:padic-bernoulli-factor,
  lem:odd-index-to-k}
For $p$ odd,
\[
  p\mid h^-(K)
  \quad\Longleftrightarrow\quad
  \exists k,\ 1\le k,\ 2k\le p-3,\quad
  p\mid\operatorname{num}(B_{2k}).
\]
\end{theorem}
\begin{proof}\leanok
By \cref{thm:hminus-bernoulli-mod-p}, $h^-$ is congruent modulo $p$ to the
finite product of the $p$-adic integers $-\tfrac12 B_{j+1}/(j+1)$ over odd
$j < p-2$.  The factor $-\tfrac12$ and the denominator $j+1$ are $p$-adic
units in this range, so by \cref{lem:padic-bernoulli-factor} each factor is
a non-unit exactly when $p$ divides $\operatorname{num}(B_{j+1})$.  A
product of $p$-adic integers lies in $p\Z_p$ exactly when one of the
factors does, so $p \mid h^-$ holds precisely when some numerator in the
range is divisible by $p$.  Finally \cref{lem:odd-index-to-k} converts the
odd index $j$ into the classical range $1\le k$, $2k\le p-3$ with
$2k = j+1$.
\end{proof}

\begin{theorem}\label{thm:hminus-nondiv-to-bernoulli-nondiv}
  \lean{KummerCriterion.bernoulli_nonzero_of_not_dvd_hMinus}\leanok
  \uses{thm:hminus-bernoulli-final}
If $p\nmid h^-(K)$, then for every $k$ with $1\le k$ and $2k\le p-3$,
\[
  p\nmid\operatorname{num}(B_{2k}).
\]
\end{theorem}
\begin{proof}\leanok
Contrapositive of one direction of \cref{thm:hminus-bernoulli-final}: a
numerator divisible by $p$ in the range would force $p\mid h^-(K)$.
\end{proof}
